import React, { useState, useEffect } from "react";
import { base44 } from "@/api/base44Client";
import { Button } from "@/components/ui/button";
import { Input } from "@/components/ui/input";
import { Label } from "@/components/ui/label";
import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
import { Badge } from "@/components/ui/badge";
import { User, Heart, AlertCircle, Leaf, Plus, X, LogOut, Activity } from "lucide-react";
import { motion } from "framer-motion";
import { useNavigate } from "react-router-dom";
import { createPageUrl } from "@/utils";

const commonAllergies = [
  "Amendoim", "Castanhas", "Nozes", "Amêndoas", "Leite", "Ovos", "Soja", 
  "Trigo", "Glúten", "Peixe", "Frutos do Mar", "Camarão", "Gergelim", 
  "Mostarda", "Sulfitos", "Aipo", "Crustáceos", "Moluscos", "Lupino", 
  "Milho", "Kiwi", "Banana", "Abacate", "Tomate", "Frutas Cítricas", 
  "Morango", "Chocolate", "Café", "Mel", "Açúcar", "Lactose", 
  "Frutose", "Corantes", "Conservantes", "Glutamato"
];

const commonHealthConditions = [
  "Diabetes Tipo 1", "Diabetes Tipo 2", "Hipertensão", "Colesterol Alto",
  "Doença Celíaca", "Intolerância à Lactose", "Síndrome do Intestino Irritável",
  "Doença de Crohn", "Gastrite", "Refluxo", "Úlcera", "Hipotireoidismo",
  "Hipertireoidismo", "Anemia", "Obesidade", "Problemas Renais",
  "Problemas Cardíacos", "Asma", "Artrite", "Osteoporose", "Gota",
  "Hiperuricemia", "Triglicerídeos Alto", "Pré-diabetes", "Resistência à Insulina"
];

const commonDietary = [
  "Vegano", "Vegetariano", "Ovolactovegetariano", "Pescetariano",
  "Sem Glúten", "Sem Lactose", "Sem Açúcar", "Baixo Carboidrato",
  "Low Carb", "Cetogênica", "Paleo", "Kosher", "Halal", 
  "Orgânico", "Sem Conservantes", "Sem Corantes", "Natural",
  "Dieta Mediterrânea", "DASH", "Flexitariana", "Crudívora"
];

const commonSustainabilityPriorities = [
  "Embalagem Ecológica", "Embalagem Reciclável", "Embalagem Biodegradável",
  "Carbono Neutro", "Emissões Reduzidas", "Comércio Justo", "Fair Trade",
  "Orgânico", "Certificado Orgânico", "Local", "Produção Local",
  "Livre de Crueldade", "Cruelty-Free", "Não Testado em Animais",
  "Vegano (Produto)", "Sustentável", "Energia Renovável", "Zero Resíduos",
  "Ingredientes Naturais", "Sem Plástico", "Reciclado", "Responsabilidade Social",
  "Agricultura Sustentável", "Pesca Sustentável", "FSC Certificado", "Rainforest Alliance"
];

export default function Profile() {
  const navigate = useNavigate();
  const [user, setUser] = useState(null);
  const [fullName, setFullName] = useState("");
  const [allergies, setAllergies] = useState([]);
  const [healthConditions, setHealthConditions] = useState([]);
  const [dietaryPreferences, setDietaryPreferences] = useState([]);
  const [sustainabilityPriorities, setSustainabilityPriorities] = useState([]);
  const [dailyCalorieTarget, setDailyCalorieTarget] = useState("");
  const [dailyProteinTarget, setDailyProteinTarget] = useState("");
  const [dailyCarbsTarget, setDailyCarbsTarget] = useState("");
  const [dailyFatsTarget, setDailyFatsTarget] = useState("");
  const [notificationEmail, setNotificationEmail] = useState("");
  const [newAllergy, setNewAllergy] = useState("");
  const [newCondition, setNewCondition] = useState("");
  const [saving, setSaving] = useState(false);

  useEffect(() => {
    const loadUser = async () => {
      try {
        const currentUser = await base44.auth.me();
        setUser(currentUser);
        setFullName(currentUser.full_name || "");
        setAllergies(currentUser.allergies || []);
        setHealthConditions(currentUser.health_conditions || []);
        setDietaryPreferences(currentUser.dietary_preferences || []);
        setSustainabilityPriorities(currentUser.sustainability_priorities || []);
        setDailyCalorieTarget(currentUser.daily_calorie_target || "");
        setDailyProteinTarget(currentUser.daily_protein_target || "");
        setDailyCarbsTarget(currentUser.daily_carbs_target || "");
        setDailyFatsTarget(currentUser.daily_fats_target || "");
        setNotificationEmail(currentUser.notification_email || currentUser.email);
      } catch (error) {
        console.error("Error loading user:", error);
      }
    };

    loadUser();
  }, []);

  const addItem = (list, setList, item) => {
    if (item && !list.includes(item)) {
      setList([...list, item]);
    }
  };

  const removeItem = (list, setList, item) => {
    setList(list.filter(i => i !== item));
  };

  const handleSave = async () => {
    setSaving(true);
    try {
      await base44.auth.updateMe({
        full_name: fullName,
        allergies,
        health_conditions: healthConditions,
        dietary_preferences: dietaryPreferences,
        sustainability_priorities: sustainabilityPriorities,
        daily_calorie_target: dailyCalorieTarget ? Number(dailyCalorieTarget) : undefined,
        daily_protein_target: dailyProteinTarget ? Number(dailyProteinTarget) : undefined,
        daily_carbs_target: dailyCarbsTarget ? Number(dailyCarbsTarget) : undefined,
        daily_fats_target: dailyFatsTarget ? Number(dailyFatsTarget) : undefined,
        notification_email: notificationEmail || undefined,
        profile_completed: true
      });
      navigate(createPageUrl("Home"));
    } catch (error) {
      console.error("Error saving profile:", error);
    }
    setSaving(false);
  };

  const handleLogout = () => {
    base44.auth.logout();
  };

  if (!user) return null;

  return (
    <div className="min-h-screen px-4 py-8 pb-24">
      <div className="max-w-4xl mx-auto">
        <motion.div
          initial={{ opacity: 0, y: 20 }}
          animate={{ opacity: 1, y: 0 }}
          transition={{ duration: 0.6 }}
        >
          <div className="flex items-center justify-between mb-8">
            <div>
              <h1 className="text-3xl md:text-4xl font-bold text-gray-800 mb-2">Seu Perfil</h1>
              <p className="text-gray-600">Configure suas preferências pessoais e metas nutricionais</p>
            </div>
            <Button
              variant="outline"
              onClick={handleLogout}
              className="flex items-center gap-2"
            >
              <LogOut className="w-4 h-4" />
              Sair
            </Button>
          </div>

          <div className="space-y-6">
            {/* Basic Info */}
            <Card className="border-none shadow-lg">
              <CardHeader>
                <CardTitle className="flex items-center gap-2">
                  <User className="w-5 h-5 text-purple-600" />
                  Informações Básicas
                </CardTitle>
              </CardHeader>
              <CardContent className="space-y-4">
                <div>
                  <Label htmlFor="fullName">Nome Completo *</Label>
                  <Input
                    id="fullName"
                    value={fullName}
                    onChange={(e) => setFullName(e.target.value)}
                    placeholder="Digite seu nome completo"
                    className="mt-1"
                  />
                </div>
                <div>
                  <Label>E-mail de Cadastro</Label>
                  <Input value={user.email} disabled className="mt-1 bg-gray-50" />
                </div>
                <div>
                  <Label htmlFor="notificationEmail">E-mail para Notificações (Opcional)</Label>
                  <Input
                    id="notificationEmail"
                    type="email"
                    value={notificationEmail}
                    onChange={(e) => setNotificationEmail(e.target.value)}
                    placeholder="Digite um e-mail alternativo para receber notificações"
                    className="mt-1"
                  />
                  <p className="text-xs text-gray-500 mt-1">
                    Deixe em branco para usar o e-mail de cadastro
                  </p>
                </div>
              </CardContent>
            </Card>

            {/* Nutritional Goals */}
            <Card className="border-none shadow-lg">
              <CardHeader>
                <CardTitle className="flex items-center gap-2">
                  <Activity className="w-5 h-5 text-orange-600" />
                  Metas Nutricionais Diárias (Opcional)
                </CardTitle>
              </CardHeader>
              <CardContent>
                <p className="text-sm text-gray-600 mb-4">
                  Configure suas metas para comparação com produtos escaneados
                </p>
                <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
                  <div>
                    <Label htmlFor="calories">Calorias (kcal)</Label>
                    <Input
                      id="calories"
                      type="number"
                      value={dailyCalorieTarget}
                      onChange={(e) => setDailyCalorieTarget(e.target.value)}
                      placeholder="Ex: 2000"
                      className="mt-1"
                    />
                  </div>
                  <div>
                    <Label htmlFor="protein">Proteínas (g)</Label>
                    <Input
                      id="protein"
                      type="number"
                      value={dailyProteinTarget}
                      onChange={(e) => setDailyProteinTarget(e.target.value)}
                      placeholder="Ex: 150"
                      className="mt-1"
                    />
                  </div>
                  <div>
                    <Label htmlFor="carbs">Carboidratos (g)</Label>
                    <Input
                      id="carbs"
                      type="number"
                      value={dailyCarbsTarget}
                      onChange={(e) => setDailyCarbsTarget(e.target.value)}
                      placeholder="Ex: 250"
                      className="mt-1"
                    />
                  </div>
                  <div>
                    <Label htmlFor="fats">Gorduras (g)</Label>
                    <Input
                      id="fats"
                      type="number"
                      value={dailyFatsTarget}
                      onChange={(e) => setDailyFatsTarget(e.target.value)}
                      placeholder="Ex: 65"
                      className="mt-1"
                    />
                  </div>
                </div>
              </CardContent>
            </Card>

            {/* Allergies */}
            <Card className="border-none shadow-lg">
              <CardHeader>
                <CardTitle className="flex items-center gap-2">
                  <AlertCircle className="w-5 h-5 text-red-600" />
                  Alergias e Intolerâncias
                </CardTitle>
              </CardHeader>
              <CardContent className="space-y-4">
                <div className="flex flex-wrap gap-2 max-h-64 overflow-y-auto p-2">
                  {commonAllergies.map((allergy) => (
                    <Badge
                      key={allergy}
                      variant={allergies.includes(allergy) ? "default" : "outline"}
                      className={`cursor-pointer transition-all ${
                        allergies.includes(allergy)
                          ? "bg-red-500 hover:bg-red-600"
                          : "hover:bg-red-50"
                      }`}
                      onClick={() =>
                        allergies.includes(allergy)
                          ? removeItem(allergies, setAllergies, allergy)
                          : addItem(allergies, setAllergies, allergy)
                      }
                    >
                      {allergy}
                      {allergies.includes(allergy) && <X className="w-3 h-3 ml-1" />}
                    </Badge>
                  ))}
                </div>
                <div className="flex gap-2">
                  <Input
                    placeholder="Adicionar alergia personalizada"
                    value={newAllergy}
                    onChange={(e) => setNewAllergy(e.target.value)}
                    onKeyPress={(e) => {
                      if (e.key === "Enter") {
                        addItem(allergies, setAllergies, newAllergy);
                        setNewAllergy("");
                      }
                    }}
                  />
                  <Button
                    variant="outline"
                    onClick={() => {
                      addItem(allergies, setAllergies, newAllergy);
                      setNewAllergy("");
                    }}
                  >
                    <Plus className="w-4 h-4" />
                  </Button>
                </div>
                {allergies.filter(a => !commonAllergies.includes(a)).length > 0 && (
                  <div className="flex flex-wrap gap-2">
                    {allergies.filter(a => !commonAllergies.includes(a)).map((allergy) => (
                      <Badge key={allergy} className="bg-red-500">
                        {allergy}
                        <X
                          className="w-3 h-3 ml-1 cursor-pointer"
                          onClick={() => removeItem(allergies, setAllergies, allergy)}
                        />
                      </Badge>
                    ))}
                  </div>
                )}
              </CardContent>
            </Card>

            {/* Health Conditions */}
            <Card className="border-none shadow-lg">
              <CardHeader>
                <CardTitle className="flex items-center gap-2">
                  <Heart className="w-5 h-5 text-pink-600" />
                  Condições de Saúde
                </CardTitle>
              </CardHeader>
              <CardContent className="space-y-4">
                <div className="flex flex-wrap gap-2 max-h-64 overflow-y-auto p-2">
                  {commonHealthConditions.map((condition) => (
                    <Badge
                      key={condition}
                      variant={healthConditions.includes(condition) ? "default" : "outline"}
                      className={`cursor-pointer transition-all ${
                        healthConditions.includes(condition)
                          ? "bg-pink-500 hover:bg-pink-600"
                          : "hover:bg-pink-50"
                      }`}
                      onClick={() =>
                        healthConditions.includes(condition)
                          ? removeItem(healthConditions, setHealthConditions, condition)
                          : addItem(healthConditions, setHealthConditions, condition)
                      }
                    >
                      {condition}
                      {healthConditions.includes(condition) && <X className="w-3 h-3 ml-1" />}
                    </Badge>
                  ))}
                </div>
                <div className="flex gap-2">
                  <Input
                    placeholder="Adicionar condição personalizada"
                    value={newCondition}
                    onChange={(e) => setNewCondition(e.target.value)}
                    onKeyPress={(e) => {
                      if (e.key === "Enter") {
                        addItem(healthConditions, setHealthConditions, newCondition);
                        setNewCondition("");
                      }
                    }}
                  />
                  <Button
                    variant="outline"
                    onClick={() => {
                      addItem(healthConditions, setHealthConditions, newCondition);
                      setNewCondition("");
                    }}
                  >
                    <Plus className="w-4 h-4" />
                  </Button>
                </div>
                {healthConditions.filter(c => !commonHealthConditions.includes(c)).length > 0 && (
                  <div className="flex flex-wrap gap-2">
                    {healthConditions.filter(c => !commonHealthConditions.includes(c)).map((condition) => (
                      <Badge key={condition} className="bg-pink-500">
                        {condition}
                        <X
                          className="w-3 h-3 ml-1 cursor-pointer"
                          onClick={() => removeItem(healthConditions, setHealthConditions, condition)}
                        />
                      </Badge>
                    ))}
                  </div>
                )}
              </CardContent>
            </Card>

            {/* Dietary Preferences */}
            <Card className="border-none shadow-lg">
              <CardHeader>
                <CardTitle className="flex items-center gap-2">
                  <Leaf className="w-5 h-5 text-green-600" />
                  Preferências Alimentares
                </CardTitle>
              </CardHeader>
              <CardContent>
                <div className="flex flex-wrap gap-2 max-h-64 overflow-y-auto p-2">
                  {commonDietary.map((diet) => (
                    <Badge
                      key={diet}
                      variant={dietaryPreferences.includes(diet) ? "default" : "outline"}
                      className={`cursor-pointer transition-all ${
                        dietaryPreferences.includes(diet)
                          ? "bg-green-500 hover:bg-green-600"
                          : "hover:bg-green-50"
                      }`}
                      onClick={() =>
                        dietaryPreferences.includes(diet)
                          ? removeItem(dietaryPreferences, setDietaryPreferences, diet)
                          : addItem(dietaryPreferences, setDietaryPreferences, diet)
                      }
                    >
                      {diet}
                      {dietaryPreferences.includes(diet) && <X className="w-3 h-3 ml-1" />}
                    </Badge>
                  ))}
                </div>
              </CardContent>
            </Card>

            {/* Sustainability Priorities */}
            <Card className="border-none shadow-lg">
              <CardHeader>
                <CardTitle className="flex items-center gap-2">
                  <Leaf className="w-5 h-5 text-purple-600" />
                  Prioridades de Sustentabilidade
                </CardTitle>
              </CardHeader>
              <CardContent>
                <div className="flex flex-wrap gap-2 max-h-64 overflow-y-auto p-2">
                  {commonSustainabilityPriorities.map((priority) => (
                    <Badge
                      key={priority}
                      variant={sustainabilityPriorities.includes(priority) ? "default" : "outline"}
                      className={`cursor-pointer transition-all ${
                        sustainabilityPriorities.includes(priority)
                          ? "bg-purple-500 hover:bg-purple-600"
                          : "hover:bg-purple-50"
                      }`}
                      onClick={() =>
                        sustainabilityPriorities.includes(priority)
                          ? removeItem(sustainabilityPriorities, setSustainabilityPriorities, priority)
                          : addItem(sustainabilityPriorities, setSustainabilityPriorities, priority)
                      }
                    >
                      {priority}
                      {sustainabilityPriorities.includes(priority) && <X className="w-3 h-3 ml-1" />}
                    </Badge>
                  ))}
                </div>
              </CardContent>
            </Card>

            {/* Save Button */}
            <Button
              onClick={handleSave}
              disabled={saving || !fullName}
              className="w-full h-14 text-lg bg-gradient-to-r from-green-600 to-purple-600 hover:from-green-700 hover:to-purple-700"
            >
              {saving ? "Salvando..." : "Salvar Perfil"}
            </Button>
          </div>
        </motion.div>
      </div>
    </div>
  );
}
