import React, { useState, useEffect } from "react";
import { base44 } from "@/api/base44Client";
import { useNavigate } from "react-router-dom";
import { createPageUrl } from "@/utils";
import { Scan, Package, Leaf, TrendingUp, Shield, Users, Search } from "lucide-react";
import { Button } from "@/components/ui/button";
import { Input } from "@/components/ui/input";
import { Card, CardContent } from "@/components/ui/card";
import { motion } from "framer-motion";

export default function Home() {
  const navigate = useNavigate();
  const [user, setUser] = useState(null);
  const [searchQuery, setSearchQuery] = useState("");
  const [stats, setStats] = useState({ pantryItems: 0, sustainability: 0 });
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    const loadUser = async () => {
      try {
        const isAuthenticated = await base44.auth.isAuthenticated();
        if (!isAuthenticated) {
          navigate(createPageUrl("Welcome"));
          return;
        }

        const currentUser = await base44.auth.me();
        setUser(currentUser);
        
        // Check if profile is completed
        if (!currentUser.profile_completed) {
          navigate(createPageUrl("Profile"));
          return;
        }

        // Load stats
        const items = await base44.entities.PantryItem.filter({ created_by: currentUser.email });
        const avgSustainability = items.length > 0
          ? items.reduce((sum, item) => sum + (item.sustainability_score || 0), 0) / items.length
          : 0;
        
        setStats({
          pantryItems: items.length,
          sustainability: Math.round(avgSustainability)
        });
      } catch (error) {
        navigate(createPageUrl("Welcome"));
      } finally {
        setLoading(false);
      }
    };

    loadUser();
  }, [navigate]);

  const handleSearch = () => {
    if (searchQuery.trim()) {
      navigate(createPageUrl("Scanner") + `?search=${encodeURIComponent(searchQuery)}`);
    }
  };

  const features = [
    {
      icon: Scan,
      title: "Escaneamento Inteligente",
      description: "Escaneie produtos instantaneamente para análise completa de sustentabilidade",
      color: "from-green-500 to-emerald-600"
    },
    {
      icon: Shield,
      title: "Avaliação Rigorosa",
      description: "Conformidade governamental, avaliações de funcionários e métricas de sustentabilidade",
      color: "from-blue-500 to-purple-600"
    },
    {
      icon: Package,
      title: "Despensa Inteligente",
      description: "Acompanhe datas de validade e gerencie seu estoque sustentável",
      color: "from-amber-500 to-orange-600"
    },
    {
      icon: Leaf,
      title: "Foco Ecológico",
      description: "Encontre produtos que combinam com seus valores e necessidades de saúde",
      color: "from-teal-500 to-green-600"
    }
  ];

  if (loading) {
    return (
      <div className="min-h-screen flex items-center justify-center">
        <div className="text-center">
          <div className="w-16 h-16 mx-auto mb-4 bg-gradient-to-br from-green-600 to-purple-600 rounded-full animate-pulse"></div>
          <p className="text-gray-600">Carregando...</p>
        </div>
      </div>
    );
  }

  return (
    <div className="min-h-screen">
      {/* Hero Section */}
      <section className="relative overflow-hidden px-4 py-12 md:py-20">
        <div className="absolute inset-0 bg-gradient-to-br from-green-100 via-purple-50 to-amber-100 opacity-50" />
        <div className="absolute inset-0">
          <div className="absolute top-20 left-10 w-72 h-72 bg-purple-300 rounded-full mix-blend-multiply filter blur-xl opacity-20 animate-blob" />
          <div className="absolute top-40 right-10 w-72 h-72 bg-green-300 rounded-full mix-blend-multiply filter blur-xl opacity-20 animate-blob animation-delay-2000" />
          <div className="absolute bottom-20 left-1/2 w-72 h-72 bg-amber-300 rounded-full mix-blend-multiply filter blur-xl opacity-20 animate-blob animation-delay-4000" />
        </div>

        <div className="relative max-w-7xl mx-auto">
          <motion.div
            initial={{ opacity: 0, y: 20 }}
            animate={{ opacity: 1, y: 0 }}
            transition={{ duration: 0.6 }}
            className="text-center mb-12"
          >
            <h1 className="text-4xl md:text-6xl lg:text-7xl font-bold mb-6">
              <span className="bg-gradient-to-r from-green-700 via-green-600 to-purple-600 bg-clip-text text-transparent">
                Descubra Produtos
              </span>
              <br />
              <span className="text-gray-800">Sustentáveis</span>
            </h1>
            <p className="text-lg md:text-xl text-gray-600 max-w-2xl mx-auto mb-8">
              Faça escolhas informadas com avaliação rigorosa de produtos baseada em suas necessidades de saúde e valores de sustentabilidade
            </p>

            {/* Search Bar */}
            <div className="max-w-2xl mx-auto">
              <Card className="border-none shadow-2xl bg-white/80 backdrop-blur-sm">
                <CardContent className="p-4">
                  <div className="flex gap-2">
                    <Input
                      placeholder="Buscar produtos ou marcas..."
                      value={searchQuery}
                      onChange={(e) => setSearchQuery(e.target.value)}
                      onKeyPress={(e) => e.key === "Enter" && handleSearch()}
                      className="flex-1 h-14 text-lg border-none bg-gray-50"
                    />
                    <Button
                      onClick={handleSearch}
                      className="h-14 px-8 bg-gradient-to-r from-green-600 to-purple-600 hover:from-green-700 hover:to-purple-700"
                    >
                      <Search className="w-5 h-5" />
                    </Button>
                  </div>
                </CardContent>
              </Card>
            </div>
          </motion.div>

          {/* Quick Stats */}
          {user && (
            <motion.div
              initial={{ opacity: 0, y: 20 }}
              animate={{ opacity: 1, y: 0 }}
              transition={{ duration: 0.6, delay: 0.2 }}
              className="grid grid-cols-1 md:grid-cols-2 gap-6 max-w-2xl mx-auto mb-12"
            >
              <Card className="border-none shadow-lg bg-gradient-to-br from-green-500 to-emerald-600 text-white">
                <CardContent className="p-6">
                  <Package className="w-8 h-8 mb-2 opacity-80" />
                  <p className="text-3xl font-bold">{stats.pantryItems}</p>
                  <p className="text-green-100">Produtos na Despensa</p>
                </CardContent>
              </Card>
              <Card className="border-none shadow-lg bg-gradient-to-br from-purple-500 to-purple-700 text-white">
                <CardContent className="p-6">
                  <TrendingUp className="w-8 h-8 mb-2 opacity-80" />
                  <p className="text-3xl font-bold">{stats.sustainability}%</p>
                  <p className="text-purple-100">Pontuação Média de Sustentabilidade</p>
                </CardContent>
              </Card>
            </motion.div>
          )}

          {/* Action Buttons */}
          <motion.div
            initial={{ opacity: 0, y: 20 }}
            animate={{ opacity: 1, y: 0 }}
            transition={{ duration: 0.6, delay: 0.3 }}
            className="flex flex-col sm:flex-row gap-4 justify-center"
          >
            <Button
              onClick={() => navigate(createPageUrl("Scanner"))}
              size="lg"
              className="h-14 px-8 text-lg bg-gradient-to-r from-green-600 to-emerald-600 hover:from-green-700 hover:to-emerald-700 shadow-lg"
            >
              <Scan className="w-5 h-5 mr-2" />
              Começar a Escanear
            </Button>
            <Button
              onClick={() => navigate(createPageUrl("Pantry"))}
              size="lg"
              variant="outline"
              className="h-14 px-8 text-lg border-2 border-purple-600 text-purple-600 hover:bg-purple-50"
            >
              <Package className="w-5 h-5 mr-2" />
              Ver Despensa
            </Button>
          </motion.div>
        </div>
      </section>

      {/* Features Section */}
      <section className="px-4 py-16 bg-white">
        <div className="max-w-7xl mx-auto">
          <motion.div
            initial={{ opacity: 0 }}
            whileInView={{ opacity: 1 }}
            transition={{ duration: 0.6 }}
            viewport={{ once: true }}
            className="text-center mb-12"
          >
            <h2 className="text-3xl md:text-4xl font-bold mb-4 text-gray-800">
              Por que escolher o EcoBags?
            </h2>
            <p className="text-lg text-gray-600 max-w-2xl mx-auto">
              Avaliação completa de produtos que não deixa nada ao acaso
            </p>
          </motion.div>

          <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-4 gap-6">
            {features.map((feature, index) => (
              <motion.div
                key={index}
                initial={{ opacity: 0, y: 20 }}
                whileInView={{ opacity: 1, y: 0 }}
                transition={{ duration: 0.6, delay: index * 0.1 }}
                viewport={{ once: true }}
              >
                <Card className="h-full border-none shadow-lg hover:shadow-xl transition-shadow duration-300">
                  <CardContent className="p-6">
                    <div className={`w-14 h-14 rounded-2xl bg-gradient-to-br ${feature.color} flex items-center justify-center mb-4`}>
                      <feature.icon className="w-7 h-7 text-white" />
                    </div>
                    <h3 className="text-xl font-bold mb-2 text-gray-800">{feature.title}</h3>
                    <p className="text-gray-600">{feature.description}</p>
                  </CardContent>
                </Card>
              </motion.div>
            ))}
          </div>
        </div>
      </section>

      {/* CTA Section */}
      <section className="px-4 py-16 bg-gradient-to-br from-green-600 to-purple-600">
        <div className="max-w-4xl mx-auto text-center text-white">
          <motion.div
            initial={{ opacity: 0, scale: 0.95 }}
            whileInView={{ opacity: 1, scale: 1 }}
            transition={{ duration: 0.6 }}
            viewport={{ once: true }}
          >
            <Leaf className="w-16 h-16 mx-auto mb-6 opacity-90" />
            <h2 className="text-3xl md:text-4xl font-bold mb-4">
              Comece sua Jornada Sustentável Hoje
            </h2>
            <p className="text-lg mb-8 text-green-100">
              Junte-se a milhares fazendo escolhas conscientes por um planeta melhor
            </p>
            <Button
              onClick={() => navigate(createPageUrl("Scanner"))}
              size="lg"
              className="h-14 px-8 text-lg bg-white text-green-700 hover:bg-gray-100"
            >
              Começar Agora
            </Button>
          </motion.div>
        </div>
      </section>

      <style>{`
        @keyframes blob {
          0%, 100% { transform: translate(0, 0) scale(1); }
          33% { transform: translate(30px, -50px) scale(1.1); }
          66% { transform: translate(-20px, 20px) scale(0.9); }
        }
        .animate-blob {
          animation: blob 7s infinite;
        }
        .animation-delay-2000 {
          animation-delay: 2s;
        }
        .animation-delay-4000 {
          animation-delay: 4s;
        }
      `}</style>
    </div>
  );
}
