const compareWithDailyTargets = (nutritionFacts, userTargets) => {
  // Mapeamento dos macronutrientes
  const macros = ['calories', 'protein', 'carbohydrates', 'fats'];
  
  const comparison = {};
  
  macros.forEach(macro => {
    const value = nutritionFacts[macro] || 0;
    const target = userTargets[\`daily_\${macro}_target\`] || 0;
    
    comparison[macro] = {
      value: value,
      target: target,
      percentage: target > 0 ? Math.round((value / target) * 100) : 0,
      status: getStatus(value, target)
    };
  });
  
  return comparison;
};

// Determina status baseado na porcentagem
const getStatus = (value, target) => {
  if (target === 0) return 'undefined';
  const percentage = (value / target) * 100;
  if (percentage <= 25) return 'low';
  if (percentage <= 75) return 'moderate';
  if (percentage <= 100) return 'good';
  return 'exceeded';
};`
