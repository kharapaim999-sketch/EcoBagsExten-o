import React from "react";
import { useQuery } from "@tanstack/react-query";
import { Card, CardContent } from "@/components/ui/card";
import { Bell, AlertCircle, Calendar, Package, CheckCircle, Clock, XCircle, Flame } from "lucide-react";
import { motion } from "framer-motion";
import { format } from "date-fns";
import { ptBR } from "date-fns/locale";
import { Link } from "react-router-dom";
import { createPageUrl } from "@/utils";
import { base44 } from "@/api/base44Client";
import { buildAlertPipeline, buildAlertStats } from "@/lib/pantryAlerts";

const severityIcon = {
  expired: XCircle,
  today: Flame,
  imminent: Clock,
  soon: Calendar,
  fresh: CheckCircle,
  unknown: AlertCircle,
};

const statCards = [
  { key: "expired", label: "Vencidos", icon: XCircle, tone: "text-red-600 dark:text-red-400", bg: "bg-red-50 dark:bg-red-900/30" },
  { key: "today", label: "Hoje", icon: Flame, tone: "text-orange-600 dark:text-orange-400", bg: "bg-orange-50 dark:bg-orange-900/30" },
  { key: "imminent", label: "Urgentes", icon: Clock, tone: "text-amber-600 dark:text-amber-400", bg: "bg-amber-50 dark:bg-amber-900/30" },
  { key: "soon", label: "Em breve", icon: Calendar, tone: "text-blue-600 dark:text-blue-400", bg: "bg-blue-50 dark:bg-blue-900/30" },
];

export default function Notifications() {
  const { data: items, isLoading } = useQuery({
    queryKey: ["pantryItems"],
    queryFn: async () => {
      const currentUser = await base44.auth.me();
      return base44.entities.PantryItem.filter({ created_by: currentUser.email });
    },
    initialData: []
  });

  const alertItems = buildAlertPipeline(items);
  const stats = buildAlertStats(items);

  return (
    <div className="min-h-screen px-4 py-8 dark:bg-gradient-to-br dark:from-gray-900 dark:via-slate-900 dark:to-gray-950">
      <div className="max-w-4xl mx-auto">
        <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }} transition={{ duration: 0.6 }}>
          {/* Header */}
          <div className="flex items-center gap-3 mb-6">
            <div className="w-12 h-12 bg-gradient-to-br from-red-500 to-orange-600 rounded-2xl flex items-center justify-center shadow-lg">
              <Bell className="w-6 h-6 text-white" />
            </div>
            <div>
              <h1 className="text-2xl md:text-3xl font-bold text-gray-800 dark:text-gray-100">Notificações</h1>
              <p className="text-gray-600 dark:text-gray-400 text-sm">
                {alertItems.length} produto{alertItems.length === 1 ? "" : "s"} precisando de atenção
              </p>
            </div>
          </div>

          {/* Stats grid */}
          {!isLoading && alertItems.length > 0 && (
            <div className="grid grid-cols-4 gap-2 mb-6">
              {statCards.map(({ key, label, icon: Icon, tone, bg }) => (
                <div key={key} className={`${bg} rounded-xl p-3 text-center`}>
                  <Icon className={`w-5 h-5 mx-auto mb-1 ${tone}`} />
                  <p className={`text-xl font-bold ${tone}`}>{stats[key] || 0}</p>
                  <p className="text-[10px] text-gray-500 dark:text-gray-400 font-medium">{label}</p>
                </div>
              ))}
            </div>
          )}

          {isLoading ? (
            <div className="space-y-4">
              {[...Array(3)].map((_, i) => (
                <Card key={i} className="animate-pulse">
                  <CardContent className="p-6"><div className="h-20 bg-gray-200 dark:bg-gray-700 rounded" /></CardContent>
                </Card>
              ))}
            </div>
          ) : alertItems.length > 0 ? (
            <div className="space-y-3">
              {alertItems.map((item, index) => {
                const Icon = severityIcon[item._severity.key] || AlertCircle;
                return (
                  <motion.div
                    key={item.id}
                    initial={{ opacity: 0, x: -20 }}
                    animate={{ opacity: 1, x: 0 }}
                    transition={{ duration: 0.4, delay: index * 0.08 }}
                  >
                    <div className={`rounded-2xl border-l-4 ${item._tone.border} ${item._tone.bg} shadow-md hover:shadow-lg transition-shadow p-4`}>
                      <div className="flex items-start gap-3">
                        <div className={`w-10 h-10 rounded-xl ${item._tone.bg} flex items-center justify-center flex-shrink-0`}>
                          <Icon className={`w-5 h-5 ${item._tone.text}`} />
                        </div>
                        <div className="flex-1 min-w-0">
                          <div className="flex items-start justify-between gap-2 mb-1">
                            <div className="min-w-0">
                              <h3 className="text-base font-bold text-gray-800 dark:text-gray-100 truncate">{item.product_name}</h3>
                              <p className="text-xs text-gray-500 dark:text-gray-400">{item.brand}</p>
                            </div>
                            <span className={`text-xs font-semibold px-2 py-0.5 rounded-full ${item._tone.badge} flex-shrink-0`}>
                              {item._severity.label}
                            </span>
                          </div>
                          <div className="flex flex-wrap items-center gap-x-4 gap-y-1 mt-2">
                            <span className="flex items-center gap-1 text-xs text-gray-500 dark:text-gray-400">
                              <Calendar className="w-3.5 h-3.5" />
                              {format(new Date(item.expiration_date), "d 'de' MMM", { locale: ptBR })}
                            </span>
                            <span className={`text-xs font-semibold ${item._tone.text}`}>{item._message}</span>
                            {item.quantity > 1 && (
                              <span className="flex items-center gap-1 text-xs text-gray-500 dark:text-gray-400">
                                <Package className="w-3.5 h-3.5" />
                                {item.quantity} un
                              </span>
                            )}
                          </div>
                        </div>
                      </div>
                    </div>
                  </motion.div>
                );
              })}
            </div>
          ) : (
            <Card className="border-none shadow-lg dark:bg-gray-800">
              <CardContent className="p-10 text-center">
                <div className="w-16 h-16 mx-auto mb-4 bg-gradient-to-br from-green-100 to-green-200 dark:from-green-900/40 dark:to-green-800/40 rounded-full flex items-center justify-center">
                  <CheckCircle className="w-8 h-8 text-green-600 dark:text-green-400" />
                </div>
                <h3 className="text-lg font-semibold text-gray-800 dark:text-gray-100 mb-1">Tudo certo!</h3>
                <p className="text-gray-600 dark:text-gray-400 text-sm mb-5">Nenhum produto vencendo em breve</p>
                <Link to={createPageUrl("Pantry")} className="inline-flex items-center gap-2 text-purple-600 dark:text-purple-400 hover:text-purple-700 font-medium text-sm">
                  <Package className="w-4 h-4" />
                  Ver despensa
                </Link>
              </CardContent>
            </Card>
          )}

          {/* Info */}
          <Card className="mt-6 border-none shadow-lg bg-gradient-to-br from-purple-50 to-blue-50 dark:from-gray-800 dark:to-gray-700">
            <CardContent className="p-5">
              <h3 className="font-semibold text-gray-800 dark:text-gray-100 mb-1.5 flex items-center gap-2 text-sm">
                <Bell className="w-4 h-4 text-purple-600" />
                Sobre os alertas
              </h3>
              <p className="text-xs text-gray-600 dark:text-gray-300">
                Monitoramos sua despensa e classificamos produtos por criticidade: vencidos, vencem hoje, urgentes (até 3 dias) e em breve (até 7 dias). Mantenha seus produtos sempre atualizados para reduzir desperdícios.
              </p>
            </CardContent>
          </Card>
        </motion.div>
      </div>
    </div>
  );
}
