// ─────────────────────────────────────────────────────────────────────────────
//  Pantry Alert Engine — functional composition pipeline
//  Pure functions composed via currying & transduction for O(n) classification.
// ─────────────────────────────────────────────────────────────────────────────

const MS_PER_DAY = 86_400_000;

// ── Algebraic helpers ──────────────────────────────────────────────────────────
const curry = (fn) => (...a) =>
  a.length >= fn.length ? fn(...a) : curry((...b) => fn(...a, ...b));

const compose =
  (...fns) =>
  (x) =>
    fns.reduceRight((acc, fn) => fn(acc), x);

const liftA2 = (f) => (a) => (b) => f(a, b);

const partition = (pred) => (xs) =>
  xs.reduce(
    (acc, x) => (pred(x) ? [[...acc[0], x], acc[1]] : [acc[0], [...acc[1], x]]),
    [[], []]
  );

// ── Time lens ──────────────────────────────────────────────────────────────────
const lens = (prop) => ({
  get: (o) => o?.[prop],
  set: (val) => (o) => ({ ...o, [prop]: val }),
});

const expirationLens = lens("expiration_date");

const daysUntil = (dateStr) => {
  if (!dateStr) return null;
  const delta = new Date(dateStr).getTime() - Date.now();
  return Math.ceil(delta / MS_PER_DAY);
};

// ── Severity taxonomy (ordered by criticality) ─────────────────────────────────
const Severity = Object.freeze({
  EXPIRED: { rank: 0, key: "expired", label: "Vencido", tone: "red" },
  TODAY: { rank: 1, key: "today", label: "Vence hoje", tone: "orange" },
  IMMINENT: { rank: 2, key: "imminent", label: "Urgente", tone: "amber" },
  SOON: { rank: 3, key: "soon", label: "Em breve", tone: "blue" },
  FRESH: { rank: 4, key: "fresh", label: "Fresco", tone: "green" },
  UNKNOWN: { rank: 5, key: "unknown", label: "Sem data", tone: "slate" },
});

const classify = (days) => {
  if (days === null) return Severity.UNKNOWN;
  if (days < 0) return Severity.EXPIRED;
  if (days === 0) return Severity.TODAY;
  if (days <= 3) return Severity.IMMINENT;
  if (days <= 7) return Severity.SOON;
  return Severity.FRESH;
};

// ── Tone → Tailwind class maps (literal strings for purge safety) ──────────────
const toneMap = {
  red: { bg: "bg-red-50 dark:bg-red-900/30", border: "border-red-200 dark:border-red-800", text: "text-red-600 dark:text-red-400", dot: "bg-red-500", badge: "bg-red-100 dark:bg-red-900/40 text-red-700 dark:text-red-300" },
  orange: { bg: "bg-orange-50 dark:bg-orange-900/30", border: "border-orange-200 dark:border-orange-800", text: "text-orange-600 dark:text-orange-400", dot: "bg-orange-500", badge: "bg-orange-100 dark:bg-orange-900/40 text-orange-700 dark:text-orange-300" },
  amber: { bg: "bg-amber-50 dark:bg-amber-900/30", border: "border-amber-200 dark:border-amber-800", text: "text-amber-600 dark:text-amber-400", dot: "bg-amber-500", badge: "bg-amber-100 dark:bg-amber-900/40 text-amber-700 dark:text-amber-300" },
  blue: { bg: "bg-blue-50 dark:bg-blue-900/30", border: "border-blue-200 dark:border-blue-800", text: "text-blue-600 dark:text-blue-400", dot: "bg-blue-500", badge: "bg-blue-100 dark:bg-blue-900/40 text-blue-700 dark:text-blue-300" },
  green: { bg: "bg-green-50 dark:bg-green-900/30", border: "border-green-200 dark:border-green-800", text: "text-green-600 dark:text-green-400", dot: "bg-green-500", badge: "bg-green-100 dark:bg-green-900/40 text-green-700 dark:text-green-300" },
  slate: { bg: "bg-slate-50 dark:bg-slate-800/40", border: "border-slate-200 dark:border-slate-700", text: "text-slate-600 dark:text-slate-400", dot: "bg-slate-400", badge: "bg-slate-100 dark:bg-slate-700 text-slate-700 dark:text-slate-300" },
};

// ── Message generator (polymorphic by severity rank) ───────────────────────────
const messageFor = (days) => {
  if (days === null) return "Data não informada";
  if (days < 0) return `Venceu há ${Math.abs(days)} dia${Math.abs(days) === 1 ? "" : "s"}`;
  if (days === 0) return "Vence hoje!";
  if (days === 1) return "Vence amanhã";
  return `Vence em ${days} dias`;
};

// ── Enrich: attach computed metadata to each item ──────────────────────────────
const enrich = (item) => {
  const days = daysUntil(expirationLens.get(item));
  const sev = classify(days);
  return {
    ...item,
    _days: days,
    _severity: sev,
    _tone: toneMap[sev.tone],
    _message: messageFor(days),
  };
};

// ── Predicates (curried for composition) ───────────────────────────────────────
const isAlertable = (item) =>
  item._days !== null && item._days <= 7;

const bySeverityRank = (a, b) => a._severity.rank - b._severity.rank;

// ── Stats aggregator (reduce-based monoid) ─────────────────────────────────────
const statReducer = (acc, item) => {
  const k = item._severity.key;
  return { ...acc, [k]: (acc[k] || 0) + 1, total: acc.total + 1 };
};

// ── Public API: pipeline composed from primitives ──────────────────────────────
export const buildAlertPipeline = (items) =>
  compose(
    (enriched) => enriched.filter(isAlertable),
    (enriched) => [...enriched].sort(bySeverityRank),
    (xs) => xs, // identity passthrough for future transducers
  )(items.map(enrich));

export const buildAlertStats = (items) =>
  items.map(enrich).reduce(statReducer, { total: 0 });

export const enrichItem = enrich;
export const classifyDays = classify;
export const daysUntilExpiration = daysUntil;

// ── Google Calendar event payload builder ─────────────────────────────────────
const toCalendarDate = (dateStr) => {
  const d = new Date(dateStr);
  return {
    date: d.toISOString().split("T")[0],
  };
};

    `Status: ${item._message}`,
    `Quantidade: ${item.quantity || 1}`,
    "",
    "Alerta gerado pelo EcoBags — gestão inteligente de despensa.",
  ].join("\n"),
  start: toCalendarDate(item.expiration_date),
  end: toCalendarDate(item.expiration_date),
  reminders: {
    useDefault: false,
    overrides: [
      { method: "popup", minutes: 60 * 24 * 3 },
      { method: "popup", minutes: 60 * 24 },
      { method: "popup", minutes: 60 * 3 },
    ],
  },
}));
