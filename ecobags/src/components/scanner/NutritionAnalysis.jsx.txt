import React from "react";
import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
import { Badge } from "@/components/ui/badge";
import { Progress } from "@/components/ui/progress";
import { 
  Flame, Beef, Wheat, Droplets, TrendingUp, TrendingDown, 
  Minus, AlertCircle, Sparkles, ArrowRight, Leaf
} from "lucide-react";
import { motion } from "framer-motion";

export default function NutritionAnalysis({ nutritionFacts, userTargets, alternatives }) {
  if (!nutritionFacts) return null;

  const defaultTargets = {
    daily_calorie_target: userTargets?.daily_calorie_target || 2000,
    daily_protein_target: userTargets?.daily_protein_target || 50,
    daily_carbs_target: userTargets?.daily_carbs_target || 300,
    daily_fats_target: userTargets?.daily_fats_target || 65
  };

  const macros = [
    {
      name: "Calorias",
      value: nutritionFacts.calories || 0,
      target: defaultTargets.daily_calorie_target,
      unit: "kcal",
      icon: Flame,
      color: "orange"
    },
    {
      name: "Proteínas",
      value: nutritionFacts.protein || 0,
      target: defaultTargets.daily_protein_target,
      unit: "g",
      icon: Beef,
      color: "red"
    },
    {
      name: "Carboidratos",
      value: nutritionFacts.carbohydrates || 0,
      target: defaultTargets.daily_carbs_target,
      unit: "g",
      icon: Wheat,
      color: "amber"
    },
    {
      name: "Gorduras",
      value: nutritionFacts.fats || 0,
      target: defaultTargets.daily_fats_target,
      unit: "g",
      icon: Droplets,
      color: "blue"
    }
  ];

  const getStatus = (value, target) => {
    if (target === 0) return { status: "indefinido", color: "gray", icon: Minus };
    const percentage = (value / target) * 100;
    if (percentage <= 15) return { status: "baixo", color: "blue", icon: TrendingDown };
    if (percentage <= 35) return { status: "moderado", color: "yellow", icon: Minus };
    if (percentage <= 50) return { status: "bom", color: "green", icon: TrendingUp };
    return { status: "alto", color: "red", icon: AlertCircle };
  };

  const getColorClasses = (color) => {
    const colors = {
      orange: { bg: "bg-orange-100", text: "text-orange-600", progress: "bg-orange-500", border: "border-orange-200" },
      red: { bg: "bg-red-100", text: "text-red-600", progress: "bg-red-500", border: "border-red-200" },
      amber: { bg: "bg-amber-100", text: "text-amber-600", progress: "bg-amber-500", border: "border-amber-200" },
      blue: { bg: "bg-blue-100", text: "text-blue-600", progress: "bg-blue-500", border: "border-blue-200" },
      green: { bg: "bg-green-100", text: "text-green-600", progress: "bg-green-500", border: "border-green-200" },
      yellow: { bg: "bg-yellow-100", text: "text-yellow-600", progress: "bg-yellow-500", border: "border-yellow-200" },
      gray: { bg: "bg-gray-100", text: "text-gray-600", progress: "bg-gray-500", border: "border-gray-200" }
    };
    return colors[color] || colors.gray;
  };

  const getStatusBadgeColor = (status) => {
    const colors = {
      baixo: "bg-blue-100 text-blue-700 border-blue-200",
      moderado: "bg-yellow-100 text-yellow-700 border-yellow-200",
      bom: "bg-green-100 text-green-700 border-green-200",
      alto: "bg-red-100 text-red-700 border-red-200",
      indefinido: "bg-gray-100 text-gray-700 border-gray-200"
    };
    return colors[status] || colors.indefinido;
  };

  return (
    <div className="space-y-6">
      {/* Nutrition Facts Card */}
      <Card className="border-none shadow-xl bg-gradient-to-br from-white to-emerald-50">
        <CardHeader>
          <CardTitle className="flex items-center gap-2">
            <div className="w-10 h-10 bg-gradient-to-br from-green-500 to-emerald-600 rounded-xl flex items-center justify-center">
              <Sparkles className="w-5 h-5 text-white" />
            </div>
            <div>
              <span className="text-xl">Análise Nutricional</span>
              {nutritionFacts.serving_size && (
                <p className="text-sm font-normal text-gray-500">
                  Por porção: {nutritionFacts.serving_size}
                </p>
              )}
            </div>
          </CardTitle>
        </CardHeader>
        <CardContent className="space-y-6">
          {/* Macros Grid */}
          <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
            {macros.map((macro, idx) => {
              const status = getStatus(macro.value, macro.target);
              const colorClasses = getColorClasses(macro.color);
              const percentage = macro.target > 0 ? Math.min((macro.value / macro.target) * 100, 100) : 0;

              return (
                <motion.div
                  key={macro.name}
                  initial={{ opacity: 0, y: 20 }}
                  animate={{ opacity: 1, y: 0 }}
                  transition={{ delay: idx * 0.1 }}
                  className={`p-4 rounded-xl border ${colorClasses.border} bg-white`}
                >
                  <div className="flex items-center justify-between mb-3">
                    <div className="flex items-center gap-3">
                      <div className={`w-10 h-10 rounded-lg ${colorClasses.bg} flex items-center justify-center`}>
                        <macro.icon className={`w-5 h-5 ${colorClasses.text}`} />
                      </div>
                      <div>
                        <h4 className="font-semibold text-gray-800">{macro.name}</h4>
                        <p className="text-2xl font-bold text-gray-900">
                          {macro.value}{macro.unit}
                        </p>
                      </div>
                    </div>
                    <Badge className={`${getStatusBadgeColor(status.status)} border`}>
                      <status.icon className="w-3 h-3 mr-1" />
                      {status.status}
                    </Badge>
                  </div>
                  
                  <div className="space-y-2">
                    <div className="flex justify-between text-sm">
                      <span className="text-gray-600 font-medium">Meta diária:</span>
                      <span className="text-gray-800 font-bold">{macro.target}{macro.unit}</span>
                    </div>
                    <div className="relative h-3 bg-gray-100 rounded-full overflow-hidden">
                      <motion.div
                        initial={{ width: 0 }}
                        animate={{ width: `${percentage}%` }}
                        transition={{ duration: 0.8, delay: idx * 0.1 }}
                        className={`h-full rounded-full ${colorClasses.progress}`}
                      />
                    </div>
                    <div className="flex items-center justify-between">
                      <p className="text-xs text-gray-500">
                        {percentage.toFixed(1)}% da meta
                      </p>
                      {macro.value > macro.target && (
                        <p className="text-xs font-semibold text-orange-600">
                          +{(macro.value - macro.target).toFixed(1)} {macro.unit} acima
                        </p>
                      )}
                      {macro.value < macro.target && (
                        <p className="text-xs font-semibold text-blue-600">
                          {(macro.target - macro.value).toFixed(1)} {macro.unit} restantes
                        </p>
                      )}
                    </div>
                  </div>
                </motion.div>
              );
            })}
          </div>

          {/* Additional Nutrients */}
          {(nutritionFacts.fiber || nutritionFacts.sodium || nutritionFacts.sugar || nutritionFacts.saturated_fat) && (
            <div className="grid grid-cols-2 md:grid-cols-4 gap-3 pt-4 border-t">
              {nutritionFacts.fiber > 0 && (
                <div className="text-center p-3 bg-green-50 rounded-lg">
                  <p className="text-lg font-bold text-green-700">{nutritionFacts.fiber}g</p>
                  <p className="text-xs text-green-600">Fibras</p>
                </div>
              )}
              {nutritionFacts.sugar > 0 && (
                <div className="text-center p-3 bg-pink-50 rounded-lg">
                  <p className="text-lg font-bold text-pink-700">{nutritionFacts.sugar}g</p>
                  <p className="text-xs text-pink-600">Açúcares</p>
                </div>
              )}
              {nutritionFacts.sodium > 0 && (
                <div className="text-center p-3 bg-purple-50 rounded-lg">
                  <p className="text-lg font-bold text-purple-700">{nutritionFacts.sodium}mg</p>
                  <p className="text-xs text-purple-600">Sódio</p>
                </div>
              )}
              {nutritionFacts.saturated_fat > 0 && (
                <div className="text-center p-3 bg-orange-50 rounded-lg">
                  <p className="text-lg font-bold text-orange-700">{nutritionFacts.saturated_fat}g</p>
                  <p className="text-xs text-orange-600">Gord. Saturada</p>
                </div>
              )}
            </div>
          )}

          {/* User Targets Note */}
          {!userTargets?.daily_calorie_target && (
            <div className="p-3 bg-amber-50 rounded-lg border border-amber-200">
              <p className="text-sm text-amber-700 flex items-center gap-2">
                <AlertCircle className="w-4 h-4" />
                Configure suas metas nutricionais no perfil para análise personalizada
              </p>
            </div>
          )}
        </CardContent>
      </Card>

      {/* Alternative Products */}
      {alternatives && alternatives.length > 0 && (
        <Card className="border-none shadow-xl bg-gradient-to-br from-white to-purple-50">
          <CardHeader>
            <CardTitle className="flex items-center gap-2">
              <div className="w-10 h-10 bg-gradient-to-br from-purple-500 to-pink-600 rounded-xl flex items-center justify-center">
                <Leaf className="w-5 h-5 text-white" />
              </div>
              <span className="text-xl">Alternativas Recomendadas</span>
            </CardTitle>
          </CardHeader>
          <CardContent>
            <div className="space-y-3">
              {alternatives.map((alt, idx) => (
                <motion.div
                  key={idx}
                  initial={{ opacity: 0, x: -20 }}
                  animate={{ opacity: 1, x: 0 }}
                  transition={{ delay: idx * 0.1 }}
                  className="p-4 bg-white rounded-xl border border-purple-100 hover:border-purple-300 hover:shadow-md transition-all"
                >
                  <div className="flex items-start justify-between gap-3">
                    <div className="flex-1">
                      <div className="flex items-center gap-2 mb-1">
                        <h4 className="font-semibold text-gray-800">{alt.name}</h4>
                        <Badge variant="outline" className="text-xs">
                          {alt.brand}
                        </Badge>
                      </div>
                      <p className="text-sm text-gray-600 mb-2">{alt.reason}</p>
                      {alt.nutrition_highlight && (
                        <Badge className="bg-green-100 text-green-700 border border-green-200">
                          <Sparkles className="w-3 h-3 mr-1" />
                          {alt.nutrition_highlight}
                        </Badge>
                      )}
                    </div>
                    <ArrowRight className="w-5 h-5 text-purple-400 flex-shrink-0 mt-1" />
                  </div>
                </motion.div>
              ))}
            </div>
          </CardContent>
        </Card>
      )}
    </div>
  );
}
