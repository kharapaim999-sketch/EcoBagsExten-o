import React, { useState } from "react";
import { base44 } from "@/api/base44Client";
import { Button } from "@/components/ui/button";
import { Badge } from "@/components/ui/badge";
import { Trash2, AlertCircle, CheckCircle, Calendar, Search, Loader2, Building2, Leaf, Users, ChevronDown, ChevronUp, Package, Tag, Layers } from "lucide-react";
import { motion, AnimatePresence } from "framer-motion";
import { format, differenceInDays } from "date-fns";
import { ptBR } from "date-fns/locale";

const categoryStyles = {
  food: { icon: "🍎", accent: "from-emerald-400 to-green-500", tint: "bg-emerald-50 dark:bg-emerald-900/20", text: "text-emerald-700 dark:text-emerald-300" },
  beverage: { icon: "🥤", accent: "from-blue-400 to-cyan-500", tint: "bg-blue-50 dark:bg-blue-900/20", text: "text-blue-700 dark:text-blue-300" },
  personal_care: { icon: "🧴", accent: "from-purple-400 to-violet-500", tint: "bg-purple-50 dark:bg-purple-900/20", text: "text-purple-700 dark:text-purple-300" },
  household: { icon: "🏠", accent: "from-amber-400 to-orange-500", tint: "bg-amber-50 dark:bg-amber-900/20", text: "text-amber-700 dark:text-amber-300" },
  other: { icon: "📦", accent: "from-slate-400 to-gray-500", tint: "bg-slate-50 dark:bg-slate-800/40", text: "text-slate-700 dark:text-slate-300" }
};

const categoryLabels = {
  food: "Alimentos",
  beverage: "Bebidas",
  personal_care: "Cuidados Pessoais",
  household: "Casa",
  other: "Outros"
};

export default function PantryItemCard({ item, onDelete }) {
  const [loadingDetails, setLoadingDetails] = useState(false);
  const [detailedInfo, setDetailedInfo] = useState(null);
  const [showDetails, setShowDetails] = useState(false);

  const cat = categoryStyles[item.category] || categoryStyles.other;
  const catLabel = categoryLabels[item.category] || item.category;

  const getExpirationStatus = () => {
    if (!item.expiration_date) return null;
    const today = new Date();
    const expDate = new Date(item.expiration_date);
    const daysUntilExpiration = differenceInDays(expDate, today);

    if (daysUntilExpiration < 0)
      return { status: "expired", label: "Vencido", color: "text-red-600 dark:text-red-400", bg: "bg-red-50 dark:bg-red-900/30 border-red-200 dark:border-red-800", dot: "bg-red-500" };
    if (daysUntilExpiration <= 3)
      return { status: "expiring-soon", label: "Vence em breve", color: "text-orange-600 dark:text-orange-400", bg: "bg-orange-50 dark:bg-orange-900/30 border-orange-200 dark:border-orange-800", dot: "bg-orange-500" };
    if (daysUntilExpiration <= 7)
      return { status: "expiring", label: "Vence esta semana", color: "text-yellow-600 dark:text-yellow-400", bg: "bg-yellow-50 dark:bg-yellow-900/30 border-yellow-200 dark:border-yellow-800", dot: "bg-yellow-500" };
    return { status: "fresh", label: "Fresco", color: "text-green-600 dark:text-green-400", bg: "bg-green-50 dark:bg-green-900/30 border-green-200 dark:border-green-800", dot: "bg-green-500" };
  };

  const expirationStatus = getExpirationStatus();

  const scoreColor = (score) => {
    if (score >= 75) return { ring: "text-green-500", bg: "bg-green-500", label: "Excelente" };
    if (score >= 50) return { ring: "text-yellow-500", bg: "bg-yellow-500", label: "Razoável" };
    if (score >= 25) return { ring: "text-orange-500", bg: "bg-orange-500", label: "Baixo" };
    return { ring: "text-red-500", bg: "bg-red-500", label: "Crítico" };
  };

  const sustain = item.sustainability_score !== undefined ? scoreColor(item.sustainability_score) : null;

  const fetchDetailedInfo = async () => {
    setLoadingDetails(true);
    try {
      const result = await base44.integrations.Core.InvokeLLM({
        prompt: `Forneça uma análise DETALHADA e COMPLETA sobre:

PRODUTO: ${item.product_name}
MARCA/EMPRESA: ${item.brand}
CATEGORIA: ${catLabel}

INSTRUÇÕES CRÍTICAS:
1. Pesquise informações ATUALIZADAS sobre a empresa e produto
2. Forneça análise COMPLETA (NÃO use reticências ou frases incompletas)
3. Seja ESPECÍFICO com dados concretos e exemplos
4. Inclua informações sobre sustentabilidade, práticas ESG, e impacto ambiental

ANÁLISE DEVE INCLUIR:
- História e perfil da empresa (fundação, evolução, presença global)
- Práticas de sustentabilidade detalhadas (ações concretas, metas, certificações)
- Análise ESG completa (ambiental, social, governança)
- Informações sobre o produto específico (ingredientes, processo de fabricação, embalagem)
- Impacto ambiental da produção e cadeia de suprimentos
- Iniciativas sociais e comunitárias da empresa
- Avaliações de consumidores e especialistas
- Comparação com concorrentes do setor
- Pontos fortes e áreas de melhoria
- Recomendações para o consumidor consciente`,
        add_context_from_internet: true,
        response_json_schema: {
          type: "object",
          properties: {
            company_overview: { type: "string", description: "História detalhada e perfil completo da empresa" },
            sustainability_practices: { type: "string", description: "Práticas de sustentabilidade detalhadas com exemplos concretos" },
            environmental_score: { type: "number", minimum: 0, maximum: 100, description: "Score ambiental 0-100" },
            social_score: { type: "number", minimum: 0, maximum: 100, description: "Score social 0-100" },
            governance_score: { type: "number", minimum: 0, maximum: 100, description: "Score de governança 0-100" },
            product_details: { type: "string", description: "Detalhes completos sobre o produto específico" },
            supply_chain_info: { type: "string", description: "Informações sobre cadeia de suprimentos e logística" },
            social_initiatives: { type: "string", description: "Iniciativas sociais e comunitárias da empresa" },
            consumer_reviews_summary: { type: "string", description: "Resumo de avaliações de consumidores e especialistas" },
            industry_comparison: { type: "string", description: "Comparação com concorrentes e posição no mercado" },
            strengths: { type: "array", items: { type: "string" }, description: "Lista de pontos fortes (3-5 itens)" },
            areas_for_improvement: { type: "array", items: { type: "string" }, description: "Lista de áreas para melhoria (3-5 itens)" },
            recommendations: { type: "string", description: "Recomendações para o consumidor consciente" },
            certifications: { type: "array", items: { type: "string" }, description: "Certificações ambientais e sociais" }
          },
          required: ["company_overview", "sustainability_practices", "environmental_score", "social_score", "governance_score"]
        }
      });
      setDetailedInfo(result);
      setShowDetails(true);
    } catch (error) {
      console.error("Error fetching detailed info:", error);
    }
    setLoadingDetails(false);
  };

  return (
    <motion.div
      layout
      initial={{ opacity: 0, y: 16 }}
      animate={{ opacity: 1, y: 0 }}
      exit={{ opacity: 0, scale: 0.95 }}
      transition={{ duration: 0.3 }}
    >
      <div className="group relative overflow-hidden rounded-2xl bg-white dark:bg-gray-800 shadow-md hover:shadow-xl transition-all duration-300 border border-gray-100 dark:border-gray-700">
        {/* Accent bar */}
        <div className={`h-1.5 bg-gradient-to-r ${cat.accent}`} />

        {/* Body */}
        <div className="p-5">
          {/* Header row */}
          <div className="flex items-start gap-3 mb-4">
            {/* Category icon */}
            <div className={`flex-shrink-0 w-12 h-12 rounded-xl ${cat.tint} flex items-center justify-center text-2xl`}>
              {cat.icon}
            </div>

            {/* Name + brand */}
            <div className="flex-1 min-w-0">
              <h3 className="text-base font-bold text-gray-800 dark:text-gray-100 leading-tight truncate">
                {item.product_name}
              </h3>
              <p className="text-sm text-gray-500 dark:text-gray-400 truncate">{item.brand}</p>
            </div>

            {/* Actions */}
            <div className="flex gap-1 flex-shrink-0">
              <button
                onClick={fetchDetailedInfo}
                disabled={loadingDetails}
                className="w-8 h-8 rounded-lg flex items-center justify-center text-gray-400 hover:text-purple-600 hover:bg-purple-50 dark:hover:bg-purple-900/30 transition-colors disabled:opacity-50"
                title="Buscar informações detalhadas"
              >
                {loadingDetails ? <Loader2 className="w-4 h-4 animate-spin" /> : <Search className="w-4 h-4" />}
              </button>
              <button
                onClick={onDelete}
                className="w-8 h-8 rounded-lg flex items-center justify-center text-gray-400 hover:text-red-500 hover:bg-red-50 dark:hover:bg-red-900/30 transition-colors"
                title="Remover item"
              >
                <Trash2 className="w-4 h-4" />
              </button>
            </div>
          </div>

          {/* Tags row */}
          <div className="flex flex-wrap items-center gap-2 mb-4">
            <span className={`inline-flex items-center gap-1 px-2.5 py-1 rounded-full text-xs font-medium ${cat.tint} ${cat.text}`}>
              <Tag className="w-3 h-3" />
              {catLabel}
            </span>
            {item.quantity > 1 && (
              <span className="inline-flex items-center gap-1 px-2.5 py-1 rounded-full text-xs font-medium bg-gray-100 dark:bg-gray-700 text-gray-600 dark:text-gray-300">
                <Layers className="w-3 h-3" />
                {item.quantity} un
              </span>
            )}
            {item.matches_preferences !== undefined && (
              <span className={`inline-flex items-center gap-1 px-2.5 py-1 rounded-full text-xs font-medium ${
                item.matches_preferences
                  ? "bg-green-100 dark:bg-green-900/30 text-green-700 dark:text-green-300"
                  : "bg-red-100 dark:bg-red-900/30 text-red-700 dark:text-red-300"
              }`}>
                {item.matches_preferences ? <CheckCircle className="w-3 h-3" /> : <AlertCircle className="w-3 h-3" />}
                {item.matches_preferences ? "Combina" : "Verificar"}
              </span>
            )}
          </div>

          {/* Sustainability score ring */}
          {sustain && (
            <div className="flex items-center gap-3 mb-4 p-3 rounded-xl bg-gray-50 dark:bg-gray-900/50">
              <div className="relative w-14 h-14 flex-shrink-0">
                <svg className="w-14 h-14 -rotate-90" viewBox="0 0 56 56">
                  <circle cx="28" cy="28" r="24" fill="none" strokeWidth="5" className="stroke-gray-200 dark:stroke-gray-700" />
                  <circle
                    cx="28" cy="28" r="24" fill="none" strokeWidth="5"
                    className={`${sustain.ring} transition-all duration-700`}
                    strokeLinecap="round"
                    strokeDasharray={`${(item.sustainability_score / 100) * 150.8} 150.8`}
                  />
                </svg>
                <div className="absolute inset-0 flex items-center justify-center">
                  <span className="text-sm font-bold text-gray-700 dark:text-gray-200">{item.sustainability_score}</span>
                </div>
              </div>
              <div className="flex-1">
                <p className="text-xs text-gray-500 dark:text-gray-400 mb-0.5">Sustentabilidade</p>
                <p className={`text-sm font-semibold ${sustain.ring}`}>{sustain.label}</p>
              </div>
              {item.evaluation_data && (item.evaluation_data.packaging_score !== undefined || item.evaluation_data.company_practices_score !== undefined) && (
                <div className="flex gap-3 text-center">
                  {item.evaluation_data.packaging_score !== undefined && (
                    <div>
                      <p className="text-lg font-bold text-gray-700 dark:text-gray-200">{item.evaluation_data.packaging_score}</p>
                      <p className="text-[10px] text-gray-400 uppercase tracking-wide">Embalagem</p>
                    </div>
                  )}
                  {item.evaluation_data.company_practices_score !== undefined && (
                    <div>
                      <p className="text-lg font-bold text-gray-700 dark:text-gray-200">{item.evaluation_data.company_practices_score}</p>
                      <p className="text-[10px] text-gray-400 uppercase tracking-wide">Práticas</p>
                    </div>
                  )}
                </div>
              )}
            </div>
          )}

          {/* Expiration */}
          {item.expiration_date && expirationStatus && (
            <div className={`flex items-center gap-2.5 px-3 py-2.5 rounded-xl border ${expirationStatus.bg} mb-1`}>
              <span className={`w-2 h-2 rounded-full ${expirationStatus.dot} flex-shrink-0`} />
              <div className="flex-1 min-w-0">
                <p className={`text-xs font-semibold ${expirationStatus.color}`}>{expirationStatus.label}</p>
                <p className="text-xs text-gray-500 dark:text-gray-400">
                  {format(new Date(item.expiration_date), "d 'de' MMM 'de' yyyy", { locale: ptBR })}
                </p>
              </div>
              <Calendar className={`w-4 h-4 ${expirationStatus.color} flex-shrink-0`} />
            </div>
          )}

          {/* Notes */}
          {item.notes && (
            <p className="text-xs text-gray-500 dark:text-gray-400 mt-3 italic line-clamp-2">
              "{item.notes}"
            </p>
          )}

          {/* Detailed Information Section */}
          <AnimatePresence>
            {detailedInfo && showDetails && (
              <motion.div
                initial={{ opacity: 0, height: 0 }}
                animate={{ opacity: 1, height: "auto" }}
                exit={{ opacity: 0, height: 0 }}
                className="mt-4 pt-4 border-t border-gray-100 dark:border-gray-700 space-y-3"
              >
                <div className="flex items-center justify-between">
                  <h4 className="font-semibold text-gray-800 dark:text-gray-100 flex items-center gap-2 text-sm">
                    <Building2 className="w-4 h-4 text-purple-600" />
                    Análise Detalhada
                  </h4>
                  <button onClick={() => setShowDetails(false)} className="text-gray-400 hover:text-gray-600 dark:hover:text-gray-200">
                    <ChevronUp className="w-4 h-4" />
                  </button>
                </div>

                {/* ESG Scores */}
                <div className="grid grid-cols-3 gap-2">
                  <div className="text-center p-2.5 bg-green-50 dark:bg-green-900/20 rounded-lg">
                    <Leaf className="w-5 h-5 text-green-600 mx-auto mb-0.5" />
                    <p className="text-xl font-bold text-green-700 dark:text-green-300">{detailedInfo.environmental_score}</p>
                    <p className="text-[10px] text-green-600 dark:text-green-400">Ambiental</p>
                  </div>
                  <div className="text-center p-2.5 bg-blue-50 dark:bg-blue-900/20 rounded-lg">
                    <Users className="w-5 h-5 text-blue-600 mx-auto mb-0.5" />
                    <p className="text-xl font-bold text-blue-700 dark:text-blue-300">{detailedInfo.social_score}</p>
                    <p className="text-[10px] text-blue-600 dark:text-blue-400">Social</p>
                  </div>
                  <div className="text-center p-2.5 bg-purple-50 dark:bg-purple-900/20 rounded-lg">
                    <Building2 className="w-5 h-5 text-purple-600 mx-auto mb-0.5" />
                    <p className="text-xl font-bold text-purple-700 dark:text-purple-300">{detailedInfo.governance_score}</p>
                    <p className="text-[10px] text-purple-600 dark:text-purple-400">Governança</p>
                  </div>
                </div>

                {/* Company Overview */}
                <div className="bg-gray-50 dark:bg-gray-900/40 rounded-lg p-3">
                  <h5 className="font-semibold text-gray-800 dark:text-gray-200 mb-1.5 text-sm">Sobre a Empresa</h5>
                  <p className="text-xs text-gray-600 dark:text-gray-300 leading-relaxed">{detailedInfo.company_overview}</p>
                </div>

                {/* Sustainability Practices */}
                <div className="bg-green-50 dark:bg-green-900/20 rounded-lg p-3">
                  <h5 className="font-semibold text-green-800 dark:text-green-300 mb-1.5 flex items-center gap-1.5 text-sm">
                    <Leaf className="w-3.5 h-3.5" />
                    Práticas de Sustentabilidade
                  </h5>
                  <p className="text-xs text-gray-600 dark:text-gray-300 leading-relaxed">{detailedInfo.sustainability_practices}</p>
                </div>

                {/* Product Details */}
                {detailedInfo.product_details && (
                  <div className="bg-purple-50 dark:bg-purple-900/20 rounded-lg p-3">
                    <h5 className="font-semibold text-purple-800 dark:text-purple-300 mb-1.5 flex items-center gap-1.5 text-sm">
                      <Package className="w-3.5 h-3.5" />
                      Detalhes do Produto
                    </h5>
                    <p className="text-xs text-gray-600 dark:text-gray-300 leading-relaxed">{detailedInfo.product_details}</p>
                  </div>
                )}

                {/* Supply Chain */}
                {detailedInfo.supply_chain_info && (
                  <div className="bg-blue-50 dark:bg-blue-900/20 rounded-lg p-3">
                    <h5 className="font-semibold text-blue-800 dark:text-blue-300 mb-1.5 text-sm">Cadeia de Suprimentos</h5>
                    <p className="text-xs text-gray-600 dark:text-gray-300 leading-relaxed">{detailedInfo.supply_chain_info}</p>
                  </div>
                )}

                {/* Social Initiatives */}
                {detailedInfo.social_initiatives && (
                  <div className="bg-indigo-50 dark:bg-indigo-900/20 rounded-lg p-3">
                    <h5 className="font-semibold text-indigo-800 dark:text-indigo-300 mb-1.5 text-sm">Iniciativas Sociais</h5>
                    <p className="text-xs text-gray-600 dark:text-gray-300 leading-relaxed">{detailedInfo.social_initiatives}</p>
                  </div>
                )}

                {/* Strengths and Improvements */}
                <div className="grid grid-cols-1 gap-2">
                  {detailedInfo.strengths?.length > 0 && (
                    <div className="bg-green-50 dark:bg-green-900/20 rounded-lg p-3">
                      <h5 className="font-semibold text-green-800 dark:text-green-300 mb-1.5 text-xs">✓ Pontos Fortes</h5>
                      <ul className="space-y-1">
                        {detailedInfo.strengths.map((strength, idx) => (
                          <li key={idx} className="text-xs text-gray-600 dark:text-gray-300 flex items-start gap-1.5">
                            <span className="text-green-600 mt-0.5">•</span>
                            <span>{strength}</span>
                          </li>
                        ))}
                      </ul>
                    </div>
                  )}
                  {detailedInfo.areas_for_improvement?.length > 0 && (
                    <div className="bg-orange-50 dark:bg-orange-900/20 rounded-lg p-3">
                      <h5 className="font-semibold text-orange-800 dark:text-orange-300 mb-1.5 text-xs">⚠ Áreas de Melhoria</h5>
                      <ul className="space-y-1">
                        {detailedInfo.areas_for_improvement.map((area, idx) => (
                          <li key={idx} className="text-xs text-gray-600 dark:text-gray-300 flex items-start gap-1.5">
                            <span className="text-orange-600 mt-0.5">•</span>
                            <span>{area}</span>
                          </li>
                        ))}
                      </ul>
                    </div>
                  )}
                </div>

                {/* Consumer Reviews */}
                {detailedInfo.consumer_reviews_summary && (
                  <div className="bg-amber-50 dark:bg-amber-900/20 rounded-lg p-3">
                    <h5 className="font-semibold text-amber-800 dark:text-amber-300 mb-1.5 text-sm">Avaliações de Consumidores</h5>
                    <p className="text-xs text-gray-600 dark:text-gray-300 leading-relaxed">{detailedInfo.consumer_reviews_summary}</p>
                  </div>
                )}

                {/* Industry Comparison */}
                {detailedInfo.industry_comparison && (
                  <div className="bg-slate-50 dark:bg-slate-800/40 rounded-lg p-3">
                    <h5 className="font-semibold text-slate-800 dark:text-slate-200 mb-1.5 text-sm">Comparação com Concorrentes</h5>
                    <p className="text-xs text-gray-600 dark:text-gray-300 leading-relaxed">{detailedInfo.industry_comparison}</p>
                  </div>
                )}

                {/* Certifications */}
                {detailedInfo.certifications?.length > 0 && (
                  <div className="flex flex-wrap gap-1.5">
                    <span className="text-xs font-semibold text-gray-500 dark:text-gray-400">Certificações:</span>
                    {detailedInfo.certifications.map((cert, idx) => (
                      <Badge key={idx} className="bg-blue-100 dark:bg-blue-900/40 text-blue-700 dark:text-blue-300 text-xs">
                        {cert}
                      </Badge>
                    ))}
                  </div>
                )}

                {/* Recommendations */}
                {detailedInfo.recommendations && (
                  <div className="bg-purple-50 dark:bg-purple-900/20 rounded-lg p-3 border border-purple-200 dark:border-purple-800">
                    <h5 className="font-semibold text-purple-800 dark:text-purple-300 mb-1.5 text-xs">💡 Recomendações</h5>
                    <p className="text-xs text-gray-600 dark:text-gray-300 leading-relaxed">{detailedInfo.recommendations}</p>
                  </div>
                )}
              </motion.div>
            )}
          </AnimatePresence>

          {detailedInfo && !showDetails && (
            <button
              onClick={() => setShowDetails(true)}
              className="w-full mt-3 flex items-center justify-center gap-1 py-2 text-sm text-purple-600 dark:text-purple-400 hover:bg-purple-50 dark:hover:bg-purple-900/30 rounded-lg transition-colors"
            >
              <ChevronDown className="w-4 h-4" />
              Ver Análise Detalhada
            </button>
          )}
        </div>
      </div>
    </motion.div>
  );
}
