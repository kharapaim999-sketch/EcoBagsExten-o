import React from "react";
import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
import { Badge } from "@/components/ui/badge";
import { Progress } from "@/components/ui/progress";
import { 
  Leaf, Users, Shield, Award, AlertTriangle, 
  CheckCircle, XCircle, TrendingUp, Building2, Star
} from "lucide-react";
import { motion } from "framer-motion";

export default function CompanyEvaluation({ data }) {
  const getScoreColor = (score) => {
    if (score >= 80) return "text-green-600";
    if (score >= 60) return "text-yellow-600";
    if (score >= 40) return "text-orange-600";
    return "text-red-600";
  };

  const getScoreBgColor = (score) => {
    if (score >= 80) return "bg-green-100";
    if (score >= 60) return "bg-yellow-100";
    if (score >= 40) return "bg-orange-100";
    return "bg-red-100";
  };

  const getScoreLabel = (score) => {
    if (score >= 80) return "Excelente";
    if (score >= 60) return "Bom";
    if (score >= 40) return "Regular";
    return "Precisa Melhorar";
  };

  return (
    <div className="space-y-6">
      {/* Header Card */}
      <Card className="border-none shadow-xl bg-gradient-to-br from-white to-blue-50">
        <CardContent className="p-8">
          <div className="flex flex-col md:flex-row items-start md:items-center justify-between gap-4">
            <div className="flex-1">
              <div className="flex items-center gap-3 mb-2">
                <Building2 className="w-8 h-8 text-blue-600" />
                <h2 className="text-3xl font-bold text-gray-800">{data.company_name}</h2>
              </div>
              <p className="text-xl text-gray-600 mb-4">{data.sector}</p>
              <div className="flex flex-wrap gap-2">
                {data.certifications?.slice(0, 3).map((cert) => (
                  <Badge key={cert} className="bg-blue-100 text-blue-700">
                    <Award className="w-3 h-3 mr-1" />
                    {cert}
                  </Badge>
                ))}
                {data.certifications?.length > 3 && (
                  <Badge variant="outline">+{data.certifications.length - 3} certificações</Badge>
                )}
              </div>
            </div>
            <div className="text-center">
              <div className={`text-5xl font-bold ${getScoreColor(data.overall_sustainability_score)}`}>
                {data.overall_sustainability_score}
              </div>
              <p className="text-sm text-gray-500 mt-1">Pontuação ESG</p>
              <Badge className={`mt-2 ${getScoreBgColor(data.overall_sustainability_score)}`}>
                {getScoreLabel(data.overall_sustainability_score)}
              </Badge>
            </div>
          </div>
        </CardContent>
      </Card>

      {/* ESG Scores Grid */}
      <div className="grid grid-cols-1 md:grid-cols-3 gap-6">
        <ESGScoreCard
          icon={Leaf}
          title="Ambiental"
          score={data.environmental_score}
          details={data.environmental_details}
          color="green"
        />
        <ESGScoreCard
          icon={Users}
          title="Social"
          score={data.social_score}
          details={data.social_details}
          color="blue"
        />
        <ESGScoreCard
          icon={Shield}
          title="Governança"
          score={data.governance_score}
          details={data.governance_details}
          color="purple"
        />
      </div>

      {/* Employee Reviews */}
      <Card className="border-none shadow-lg">
        <CardHeader>
          <CardTitle className="flex items-center gap-2">
            <Star className="w-5 h-5 text-yellow-600" />
            Avaliações de Funcionários
          </CardTitle>
        </CardHeader>
        <CardContent>
          <div className="flex items-center gap-4 mb-4">
            <div className="text-4xl font-bold text-gray-800">
              {data.employee_rating?.toFixed(1) || "N/A"}
            </div>
            <div className="flex-1">
              <div className="flex gap-1 mb-2">
                {[1, 2, 3, 4, 5].map((star) => (
                  <Star
                    key={star}
                    className={`w-5 h-5 ${
                      star <= Math.round(data.employee_rating)
                        ? "fill-yellow-400 text-yellow-400"
                        : "text-gray-300"
                    }`}
                  />
                ))}
              </div>
              <p className="text-sm text-gray-600">Baseado em avaliações de funcionários</p>
            </div>
          </div>
          <p className="text-gray-700 mb-4">{data.employee_reviews_summary}</p>
          
          <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
            {data.positive_points?.length > 0 && (
              <div>
                <h4 className="font-semibold text-green-700 mb-2 flex items-center gap-2">
                  <CheckCircle className="w-4 h-4" />
                  Pontos Positivos
                </h4>
                <ul className="space-y-1">
                  {data.positive_points.map((point, idx) => (
                    <li key={idx} className="text-sm text-gray-600 flex items-start gap-2">
                      <span className="text-green-500 mt-1">•</span>
                      <span>{point}</span>
                    </li>
                  ))}
                </ul>
              </div>
            )}
            {data.negative_points?.length > 0 && (
              <div>
                <h4 className="font-semibold text-red-700 mb-2 flex items-center gap-2">
                  <XCircle className="w-4 h-4" />
                  Pontos de Atenção
                </h4>
                <ul className="space-y-1">
                  {data.negative_points.map((point, idx) => (
                    <li key={idx} className="text-sm text-gray-600 flex items-start gap-2">
                      <span className="text-red-500 mt-1">•</span>
                      <span>{point}</span>
                    </li>
                  ))}
                </ul>
              </div>
            )}
          </div>
        </CardContent>
      </Card>

      {/* Recent Initiatives */}
      {data.recent_initiatives?.length > 0 && (
        <Card className="border-none shadow-lg border-l-4 border-l-green-500">
          <CardHeader>
            <CardTitle className="flex items-center gap-2 text-green-700">
              <TrendingUp className="w-5 h-5" />
              Iniciativas Recentes de Sustentabilidade
            </CardTitle>
          </CardHeader>
          <CardContent>
            <ul className="space-y-3">
              {data.recent_initiatives.map((initiative, idx) => (
                <li key={idx} className="flex items-start gap-2">
                  <CheckCircle className="w-4 h-4 text-green-500 mt-1 flex-shrink-0" />
                  <span className="text-gray-700">{initiative}</span>
                </li>
              ))}
            </ul>
          </CardContent>
        </Card>
      )}

      {/* Controversies */}
      {data.controversies?.length > 0 && (
        <Card className="border-none shadow-lg border-l-4 border-l-red-500">
          <CardHeader>
            <CardTitle className="flex items-center gap-2 text-red-600">
              <AlertTriangle className="w-5 h-5" />
              Controvérsias e Problemas
            </CardTitle>
          </CardHeader>
          <CardContent>
            <ul className="space-y-3">
              {data.controversies.map((controversy, idx) => (
                <li key={idx} className="flex items-start gap-2">
                  <AlertTriangle className="w-4 h-4 text-red-500 mt-1 flex-shrink-0" />
                  <span className="text-gray-700">{controversy}</span>
                </li>
              ))}
            </ul>
          </CardContent>
        </Card>
      )}

      {/* Compliance & Comparison */}
      <div className="grid grid-cols-1 md:grid-cols-2 gap-6">
        <Card className="border-none shadow-lg">
          <CardHeader>
            <CardTitle className="flex items-center gap-2">
              <Shield className="w-5 h-5 text-blue-600" />
              Conformidade Governamental
            </CardTitle>
          </CardHeader>
          <CardContent>
            <Badge className="mb-3 bg-blue-100 text-blue-700">
              {data.compliance_status}
            </Badge>
            <p className="text-gray-700">{data.compliance_details}</p>
          </CardContent>
        </Card>

        <Card className="border-none shadow-lg">
          <CardHeader>
            <CardTitle className="flex items-center gap-2">
              <TrendingUp className="w-5 h-5 text-purple-600" />
              Comparação no Setor
            </CardTitle>
          </CardHeader>
          <CardContent>
            <p className="text-gray-700">{data.industry_comparison}</p>
          </CardContent>
        </Card>
      </div>

      {/* All Certifications */}
      {data.certifications?.length > 0 && (
        <Card className="border-none shadow-lg">
          <CardHeader>
            <CardTitle className="flex items-center gap-2">
              <Award className="w-5 h-5 text-amber-600" />
              Certificações e Selos
            </CardTitle>
          </CardHeader>
          <CardContent>
            <div className="flex flex-wrap gap-2">
              {data.certifications.map((cert) => (
                <Badge key={cert} variant="outline" className="bg-amber-50 text-amber-700 border-amber-200">
                  {cert}
                </Badge>
              ))}
            </div>
          </CardContent>
        </Card>
      )}

      {/* Recommendation */}
      <Card className="border-none shadow-xl bg-gradient-to-br from-purple-50 to-white">
        <CardHeader>
          <CardTitle className="flex items-center gap-2">
            <CheckCircle className="w-5 h-5 text-green-600" />
            Recomendação para Consumidores Conscientes
          </CardTitle>
        </CardHeader>
        <CardContent>
          <p className="text-gray-700 text-lg">{data.recommendation}</p>
        </CardContent>
      </Card>
    </div>
  );
}

function ESGScoreCard({ icon: Icon, title, score, details, color }) {
  const colorClasses = {
    green: {
      gradient: "from-green-500 to-emerald-600",
      bg: "bg-green-100",
      text: "text-green-700"
    },
    blue: {
      gradient: "from-blue-500 to-blue-600",
      bg: "bg-blue-100",
      text: "text-blue-700"
    },
    purple: {
      gradient: "from-purple-500 to-purple-600",
      bg: "bg-purple-100",
      text: "text-purple-700"
    }
  };

  const colors = colorClasses[color] || colorClasses.green;

  const getScoreColor = (s) => {
    if (s >= 80) return "from-green-500 to-emerald-600";
    if (s >= 60) return "from-yellow-500 to-orange-500";
    if (s >= 40) return "from-orange-500 to-red-500";
    return "from-red-500 to-red-700";
  };

  return (
    <motion.div
      initial={{ opacity: 0, scale: 0.95 }}
      animate={{ opacity: 1, scale: 1 }}
      transition={{ duration: 0.4 }}
    >
      <Card className="border-none shadow-lg h-full">
        <CardContent className="p-6">
          <div className="flex items-center gap-3 mb-4">
            <div className={`w-12 h-12 rounded-xl bg-gradient-to-br ${getScoreColor(score)} flex items-center justify-center`}>
              <Icon className="w-6 h-6 text-white" />
            </div>
            <div className="flex-1">
              <h3 className="font-semibold text-gray-800">{title}</h3>
              <p className="text-2xl font-bold text-gray-800">
                {score}%
              </p>
            </div>
          </div>
          <Progress value={score} className="mb-3" />
          <p className="text-sm text-gray-600">{details}</p>
        </CardContent>
      </Card>
    </motion.div>
  );
}
